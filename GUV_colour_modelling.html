Details

DESCRIPTION

Models photoreceptor responses and spectral sensitivities using theoretical visual pigment templates. It simulates UV and green photoreceptor activation curves under experimental LED illumination to analyze relative photoreceptor excitation and the effects of unequal absolute sensitivity.

INPUTS

None (self-contained theoretical model)

OUTPUTS

GUV_colour_modelling.html (rendered report with figures)

Retina-level model

Photoreceptor Sensitivity

We define a function, StavengaSpline(), to generate visual pigment template curves for invertebrates. It supports \(A1\) pigment types and calculates the \(\alpha\)-, \(\beta\)-, and \(\gamma\)-absorption bands using modified lognormal models.

# Make a spline template for a visual pigment
StavengaSpline = function(spec_range = c(300, 700), #bounds of spectrum in nanometers
                          lambda_max,#peak sensitivity
                          a_type = 'a1'){#pigment type, only a1 available currently
  wlns =  seq(from = min(spec_range),
              to = max(spec_range), 
              length.out = 1e3) #
  # modified lognormal
  Mlognorm  = function(wl,l_max, a0, a1)
  {
    x = log10(wl/l_max)
    return(
      exp(-a0*x^2 *
            (1+a1*x+3*a1^2*x^2)
      )
    )
  }
  if(a_type == 'a1')
  {
    #alpha band
    a_band = Mlognorm(wl = wlns,
                      l_max = lambda_max, 
                      a0 = 380, 
                      a1 = 6.09)
    #beta band
    b_band = 0.29*Mlognorm(wl = wlns, 
                           l_max = 340, 
                           a0 = 247, 
                           a1 = 3.59)
    #gamma band
    g_band = 1.99*Mlognorm(wl = wlns, 
                           l_max = 276, 
                           a0 = 647, 
                           a1 = 23.4)
  }else
  {stop('a2 and a3 pigments not yet implemented')}
  # N.B. Stavenga normalises to max(a.band), I normalise to function max
  r_stav  = (a_band + b_band + g_band)/
    max(a_band + b_band + g_band)
  return(    smooth.spline(x = wlns,
                           y =  r_stav)    )
}#StavengaSpline <- function(spec.range, lambda.max)

Response Function

We model the photoreceptor response (in mV) to light intensity (photons/cm²/s) using a Naka-Rushton-like function. The parameters are based on honeybee photoreceptor recordings from Kolyfetis et al. (2025). See 1Data/250327_009_Apis_MainRetina_Kolyfetis2025.pdf.

# Define a function for photoreceptor response
#params based on 250327_006 from Kolyfetis et al., 2025
PRresponse = function(
            quanta = 10^seq(from = 10, 
                            to  = 14, 
                            length.out = 21), #photons/cm2/s
            vmax = 50,#mV
            k50 = 0.03*1e14,#inflection point
            n = 1.07)#slope
  {
mV = (vmax*quanta^n)/ 
      (k50^n + quanta^n)
}
#this sets sensitivity in the following way
## rel_quanta = quanta/max(quanta)
## yy = mV/vmax
## sensitivity = k50*(yy/(1-yy))^(n^-1)

We establish parameters for the response peaks of honeybee photoreceptors (Menzel et al., 1986) and the emission peaks for the experimental ultraviolet and green LEDs, using a 20nm bandwidth range for the LED spectral distribution.

# Response peak for honeybee photoreceptors Menzel et al., 1986
lmax_u = 343
lmax_g = 543
# Emission peak for each LED
led_u = 365
led_g = 528
# A 20nm (uniform) band for each 
wlband = seq(from = -10, 
             to  = 10, 
             by = 0.1)

Sensitivity Curve Predictions

We predict the photoreceptors’ sensitivity curves across the insect-visible spectrum and specifically for the LED emission bands. We also calculate the integrated sensitivity for the green photoreceptor under green and UV LEDs to find its relative UV sensitivity (g_rel_u).

# Predict sensitivity curves across whole spectrum
u_photopig = predict(StavengaSpline(lambda_max = lmax_u))
g_photopig = predict(StavengaSpline(lambda_max = lmax_g))

# Specifically for the LEDs
#green photoreceptor
gPR_uLED = predict(StavengaSpline(lambda_max = lmax_g), x = led_u + wlband)
gPR_gLED = predict(StavengaSpline(lambda_max = lmax_g), x = led_g + wlband)
#UV photoreceptor
uPR_gLED = predict(StavengaSpline(lambda_max = lmax_u), x = led_g + wlband)
uPR_uLED = predict(StavengaSpline(lambda_max = lmax_u), x = led_u + wlband)

# integrated sensitivity
sens_gPR_uLED = integrate.xy(gPR_uLED)
sens_gPR_gLED = integrate.xy(gPR_gLED)
g_rel_u = sens_gPR_uLED/sens_gPR_gLED

Set up stimuli

  • Purpose: build LED_rspec (wl + emission curves) using Cauchy fits, convert photon flux to µmol/m²/s, create named stimulus columns (UVdimmer/UVdim/UVbright, Gdim/Gbright) and plot them.

  • Key inputs: led_u, led_g, wlband, micromol_stim vector. Uses dcauchy and plotting/polygon calls.

  • Outputs: LED_rspec data.frame and an emission plot.

  • Checks: ensure led_u/led_g/wlband exist, dcauchy is available, unit conversions correct, and intermediate columns removed intentionally (rm).

LED_rspec = data.frame(wl = seq(from = 300,
                                to = 700,
                                by = 1.0) # pavo only uses 1 nm increments
                       )
#uniform version
LED_rspec_rect = within(LED_rspec,
                   {
                     LEDu = ifelse(test = wl > min(led_u + wlband) & wl < max(led_u + wlband),
                                   yes = 1,
                                   no = 0
                                   )
                     LEDg = ifelse(test = wl > min(led_g + wlband) & wl < max(led_g + wlband),
                                   yes = 1,
                                   no = 0
                                   )
                   }
                   )

#Cauchy (typical LED emission)
LED_rspec = within(LED_rspec,
                   {
                     LEDu = dcauchy(x = wl,
                                    location = led_u,
                                    scale = diff(range(wlband))/2 # for Cauchy FWHM = 2γ
                                   )
                     LEDg = dcauchy(x = wl,
                                    location = led_g,
                                    scale = diff(range(wlband))/2 # for Cauchy FWHM = 2γ
                                   )
                   }
                   )
#convert photon flux from photons/cm2/s to micro mol/m2/s
micromol_stim <- c(dimmer = 1.2e12, dim = 4e12, bright = 4e14) *
                    1e4 * #cm2 -> m2
                    1e6 / #mol -> micromol
                    6.022e23 # Avogadro's number

LED_rspec = within(LED_rspec,
                   {
                     UVdimmer = micromol_stim['dimmer']*
                       LEDu#/ # at those wavelengths in the LED emission halfmax
                       # diff(range(wlband)) #divide by width (nm)
                     UVdim = micromol_stim['dim']*LEDu#/
                       # diff(range(wlband))
                     UVbright = micromol_stim['bright']*LEDu#/
                       # diff(range(wlband))
                     Gdim = micromol_stim['dim']*LEDg#/
                       # diff(range(wlband))
                     Gbright = micromol_stim['bright']*LEDg#/
                       # diff(range(wlband))
                     rm('LEDu', 'LEDg')
                   }
                   )
LED_rspec_rect = within(LED_rspec_rect,
                   {
                     UVdimmer = micromol_stim['dimmer']*
                       LEDu/ # at those wavelengths in the LED emission halfmax
                       diff(range(wlband)) #divide by width (nm)
                     UVdim = micromol_stim['dim']*LEDu/
                        diff(range(wlband))
                     UVbright = micromol_stim['bright']*LEDu/
                        diff(range(wlband))
                     Gdim = micromol_stim['dim']*LEDg/
                        diff(range(wlband))
                     Gbright = micromol_stim['bright']*LEDg/
                        diff(range(wlband))
                     rm('LEDu', 'LEDg')
                   }
                   )

# Open a blank frame wrapper with explicit limits 
vls = unlist(c(LED_rspec[-1], LED_rspec_rect[-1]))
mn = min(vls[vls>0])
plot(

Simulate quantum catches

• Purpose: compute photoreceptor quantum catches (pavo::vismodel), map into colour space (colspace), and compute chromatic/achromatic distances (coldist) including a rescaled run to avoid log errors. • Key inputs: LED_rspec, visual=“apis”, illum/bkg settings, scale=1e6 for brighter run. • Outputs: qcatches, bee_space, percept_distances, bee_jnd (JND dS/dL). • Checks: pass vismodel object (not colspace) to coldist when required; if you see log-scale errors, adjust scale or bkg; confirm relative/vonkries choices match experiment.

qcatches = vismodel(
  rspecdata = LED_rspec, # Wavelengths + emission intensities
  visual = "apis",              # Honeybee trichromatic photoreceptors
  achromatic = "l",             # Use Green receptor for brightness contrast
  illum = "ideal",              # These are emission spectra not reflectances
  bkg = "ideal",                # Flat baseline (dark room)
  vonkries = FALSE,             # Turn off adaptation; no background light exists
  relative = FALSE               # Estimate absolute catches
)
# Project catches into the 2D space
bee_space = colspace(vismodeldata = qcatches)#use default colour space
## Warning: Quantum catch are not relative, and have been transformed
# Calculate chromatic and achromatic distances between all stimuli
percept_distances = coldist(bee_space,
                            noise = "quantum")#minimal neural noise under dark conditions

Bezold-Brücke model

• Purpose: simulate an intensity sweep (Bezold–Brücke) for a 365 nm LED: get receptor sensitivity matrix from vismodel, map photon flux → receptor mV (PRresponse), compute JND shift (manual RNL), and plot intensity-dependent hue shift. • Key inputs: LED_rspec (UVdimmer), PRresponse function, flux_range, sens_matrix, RNL noise params (n_ratios, weber). • Outputs: ds_shifts vector and log-x plot showing predicted chromatic shift vs intensity. • Checks: PRresponse must be defined and return consistent units; sens_matrix must have correct row name (“UVdimmer”); watch numeric stability for very large flux values and ensure weber/e_param choice is appropriate.

# --- Use pavo to get the relative spectral overlap ratios ---

stimulated_spectra = as.rspec(LED_rspec[c('wl','UVdimmer')])
## wavelengths found in column 1
# Get clean, un-normalized sensitivity weights (relative = FALSE, scale = 1)
# This extracts the exact proportion of light each receptor type captures per wavelength
base_weights = vismodel(
  rspecdata = stimulated_spectra,
  visual = "apis",
  achromatic = "l",
  illum = "ideal",
  bkg = 0,
  vonkries = FALSE,
  relative = FALSE,
  scale = 1
)

# Extract just the UV (s), Blue (m), and Green (l) sensitivity matrix
sens_matrix = as.matrix(base_weights[, c("s", "m", "l")])




# --- Simulate the Bezold–Brücke intensity sweep ---
# We will scale a single 365nm UV LED across your experimental range
flux_range = 10^seq(from = 10,
                    to = 15,
                    length.out = 1e3) # photons/cm2/s
ds_shifts = numeric(length(flux_range))

# Calculate the millivolt baseline at your lowest intensity (10^10 photons)
# Multiplying the flux by the sensitivity matrix distributes the photons across the channels
phi_baseline = flux_range[1] * sens_matrix["UVdimmer", ]
mv_baseline = PRresponse(phi_baseline)

# Standard Apis RNL noise parameters
n_ratios  = c(1, 2, 6) # UV, Blue, Green receptor abundance
weber    =  0.05
e_param  = sqrt(weber^2 + 1/n_ratios)

# --- PART 4: Compute the JND color shift manually ---
for(i in 1:length(flux_range)) {
  # Calculate actual photons caught by each receptor type at this specific brightness
  phi_current = flux_range[i] * sens_matrix["UVdimmer", ]
  
  # CRITICAL PHYSIOLOGY STEP: Convert the photons into true millivolts using your function
  mv_current = PRresponse(phi_current)
  
  # In visual ecology, when using real mV responses, the signal difference 
  # is modeled as the direct subtraction of millivolts divided by noise, 
  # bypassing pavo's log step because the Naka-Rushton curves ALREADY handle the scaling compression!
  delta_signal = mv_current - mv_baseline
  
  # Manual RNL distance formula for trichromats
  num_param <- (delta_signal[2] - delta_signal[3])^2 *
    e_param[1]^2 +
    (delta_signal[3] - delta_signal[1])^2 * e_param[2]^2 +
    (delta_signal[1] - delta_signal[2])^2 * e_param[3]^2
  
  den_param <- e_param[1]^2 *
    e_param[2]^2 +
    e_param[2]^2 * e_param[3]^2 +
    e_param[3]^2 * e_param[1]^2
  
  ds_shifts[i] <- sqrt(num_param / den_param)
}

# --- Plot your Intensity-dependent Hue Shift ---
plot(
  x = NULL,
  log = 'x',
  type = "l", 
  xlab = "UV intensity (photons/cm2/s)",
  ylab = "Chromatic Distance Shift (JND)",
  main = "Predicted Bezold–Brücke Shift for 365nm LED",
  ylim = c(0,1)*2,
  xlim = range(flux_range)
)

        col = gray(0.5, 0.2),
        border = NA
        )

  y = ds_shifts,
